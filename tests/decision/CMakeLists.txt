cmake_minimum_required(VERSION 3.16)
project(geniex_decision_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
# Match sdk/CMakeLists.txt: the bundled fmt requires UTF-8 source on MSVC.
add_compile_options("$<$<C_COMPILER_ID:MSVC>:/utf-8>" "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
# Keep the test next to llama.cpp's DLLs; Windows has no rpath.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
option(GENIEX_DECISION_REFERENCE_TESTS "Compare Liquid adapters with llama.cpp systemone" OFF)
set(LLAMA_BUILD_SERVER ${GENIEX_DECISION_REFERENCE_TESTS} CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
get_filename_component(ROOT ../.. ABSOLUTE)
add_subdirectory(${ROOT}/third-party/llama.cpp llama EXCLUDE_FROM_ALL)
add_executable(decision-tests test_decision.cpp
    ${ROOT}/sdk/plugins/llama_cpp/src/decision.cpp
    ${ROOT}/sdk/plugins/llama_cpp/src/liquid-decision.cpp)
target_include_directories(decision-tests PRIVATE ${ROOT}/sdk/include ${ROOT}/sdk/plugins/llama_cpp/include)
target_compile_definitions(decision-tests PRIVATE PROJECT_SOURCE_DIR="${ROOT}")
target_link_libraries(decision-tests PRIVATE llama llama-common mtmd)
if(GENIEX_DECISION_REFERENCE_TESTS)
    target_compile_definitions(decision-tests PRIVATE GENIEX_DECISION_REFERENCE_TESTS)
    target_include_directories(decision-tests PRIVATE ${ROOT}/third-party/llama.cpp/tools/server)
    target_link_libraries(decision-tests PRIVATE llama-server-impl)
endif()
enable_testing()
add_test(NAME decision-protocols COMMAND decision-tests)
